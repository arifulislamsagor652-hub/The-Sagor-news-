export default async function handler(req, res) { const Parser = require('rss-parser'); const parser = new Parser(); const feeds = [ { name: 'আগামীর সময়', url: 'https://www.agamirshomoy.com/rss' }, { name: 'ঢাকা পোস্ট', url: 'https://www.dhakapost.com/rss' }, { name: 'সমকাল', url: 'https://samakal.com/rss' } ]; let allNews = []; for (let feed of feeds) { try { let feedData = await parser.parseURL(feed.url); feedData.items.forEach(item => { allNews.push({ title: item.title, link: item.link, source: feed.name, pubDate: item.pubDate }); }); } catch (e) {} } res.setHeader('Access-Control-Allow-Origin', '*'); res.status(200).json(allNews); }
